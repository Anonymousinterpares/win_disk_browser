    function formatSize(bytes) {
        if (bytes === 0) return '0 B';
        const k = 1024;
        const sizes = ['B', 'KB', 'MB', 'GB', 'TB', 'PB'];
        const i = Math.floor(Math.log(bytes) / Math.log(k));
        return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
    }

    function debounce(func, delay) {
        let timer;
        return function(...args) {
            clearTimeout(timer);
            timer = setTimeout(() => func.apply(this, args), delay);
        };
    }

        return {

            height: chartContainer.offsetHeight

    // Breadcrumb navigation
    function updateBreadcrumb(path) {
        breadcrumbContainer.innerHTML = '';
        if (!path) return;
        
        const parts = path.replace(/\\/g, '/').split('/').filter(p => p);
        let pathSegments = [];

        parts.forEach((part, index) => {
            pathSegments.push(part);
            
            let currentBuiltPath = pathSegments.join('\\');
            if (index === 0 && currentBuiltPath.endsWith(':')) {
                currentBuiltPath += '\\';
            }
            
            if (index > 0) {
                const separator = document.createElement('span');
                separator.className = 'breadcrumb-separator';
                separator.textContent = ' › ';
                breadcrumbContainer.appendChild(separator);
            }

            const link = document.createElement('a');
            link.className = 'breadcrumb-link';
            link.textContent = part;
            link.setAttribute('data-path', currentBuiltPath);
            
            link.addEventListener('click', (event) => {
                const targetPath = event.currentTarget.getAttribute('data-path');
                
                navStack = []; 
                if (currentView === 'sunburst') {
                    pywebview.api.get_sunburst_adaptive_view(targetPath, 4).then(data => {
                        if(data) renderView(data);
                    });
                } else {
                    currentZoom = 1.0;
                    requestTreemapView(targetPath, currentZoom);
                }
            });
            
            breadcrumbContainer.appendChild(link);
        });
    }

        let itemData = hoveredData;
        if (!itemData) {
            const rect = chartContainer.getBoundingClientRect();
            itemData = getBlockUnderMouse(x - rect.left, y - rect.top);
        }
        
        if (!itemData || !itemData.path) {
            console.log('No valid item data for context menu');
            return;
        }
    
        contextMenu.innerHTML = '';
        contextMenu.style.left = `${x}px`;
        contextMenu.style.top = `${y}px`;
        contextMenu.style.display = 'block';

        const isRealPath = itemData.path && !itemData.path.includes('\\[');
        
        // --- Core "Open Location" Feature ---
        if (isRealPath) {
            const openLocationItem = document.createElement('div');
            openLocationItem.className = 'context-menu-item';
            openLocationItem.innerHTML = '🗂️ Open Location';
            openLocationItem.onclick = () => {
                contextMenu.style.display = 'none';
                pywebview.api.open_location(itemData.path);
            };
            contextMenu.appendChild(openLocationItem);
        }

        // --- Treemap-Specific Features ---
        if (currentView === 'treemap') {
            if(isRealPath) {
                const separator = document.createElement('div');
                separator.className = 'context-menu-separator';
                contextMenu.appendChild(separator);
            }

            // "Explore This" is for directories or aggregated blocks
            if (itemData.is_dir || itemData.aggregated) {
                const exploreItem = document.createElement('div');
                exploreItem.className = 'context-menu-item';
                exploreItem.innerHTML = '🔍 Explore This';
                exploreItem.onclick = () => {
                    contextMenu.style.display = 'none';
                    spatialZoomIntoBlock(itemData);
                };
                contextMenu.appendChild(exploreItem);
            }

            // "Go Up" is always available
            const zoomOutItem = document.createElement('div');
            zoomOutItem.className = 'context-menu-item';
            zoomOutITem.innerHTML = '🔙 Go Up / Zoom Out';
            zoomOutItem.onclick = () => {
                contextMenu.style.display = 'none';
                spatialZoomOut();
            };
            contextMenu.appendChild(zoomOutItem);
        }
        
        // --- Sunburst-Specific Features ---
        if (currentView === 'sunburst' && isRealPath && itemData.is_dir) {
            const separator = document.createElement('div');
            separator.className = 'context-menu-separator';
            contextMenu.appendChild(separator);

                contextMenu.style.display = 'none';

                    if (newData) renderView(newData);
                });
            };
            contextMenu.appendChild(exploreInSunburstItem);
        }
    }

    function spatialZoomIntoBlock(blockData) {
        if (!blockData || !blockData.path) return;

        // An aggregated item is a virtual folder we can explore
        const isAggregated = blockData.aggregated === true || blockData.path.includes('\\[');
        if (isAggregated) {
            const parentPath = blockData.path.substring(0, blockData.path.lastIndexOf('\\['));
            pywebview.api.expand_aggregation(parentPath, blockData.path).then(items => {
                if (items && items.length > 0) {
                    zoomStack.push({ path: currentPath });
                    const expandedData = { name: blockData.name, path: blockData.path, value: blockData.value, children: items };
                    renderView(expandedData);
                }
            });
            return;
        }

        // Only zoom into actual directories
        if (blockData.is_dir) {
            zoomStack.push({ path: currentPath });
            requestTreemapView(blockData.path, 1.0);
        }
    }

    function spatialZoomOut() {
        if (zoomStack.length > 0) {
            const previousState = zoomStack.pop();
            requestTreemapView(previousState.path, 1.0);
        } else {
            // If stack is empty, go to parent directory as a fallback
            const isRoot = currentPath.endsWith(':\\') && currentPath.length === 3;
            if (currentPath && !isRoot) {
                let parentPath = currentPath.substring(0, currentPath.lastIndexOf('\\'));
                if (!parentPath.includes('\\')) parentPath += '\\';
                requestTreemapView(parentPath, 1.0);
            }
        }
    }

    function getBlockUnderMouse(mouseX, mouseY) {
        // This function attempts to find the data for the block under the cursor
        // by inspecting ECharts' internal structures.
        try {
            const hoveredElement = chartInstance.getZr().handler.findHover(mouseX, mouseY);
            if (hoveredElement && hoveredElement.target) {
                const target = hoveredElement.target;
                
                // The most reliable way to get data is often through an internal property
                // that holds the data index.
                for (const prop of Object.keys(target)) {
                    if (prop.startsWith('__ec_inner_')) {
                        if (target[prop] && typeof target[prop].dataIndex === 'number') {
                            const dataIndex = target[prop].dataIndex;
                            const seriesModel = chartInstance.getModel().getComponent('series', 0);
                            if (seriesModel) {
                                const rawDataItem = seriesModel.getData().getRawDataItem(dataIndex);
                                if (rawDataItem && rawDataItem.path) {
                                    return rawDataItem; // Success!
                                }
                            }
                        }
                    }
                }
            }
        } catch (e) {
            console.error('Error in getBlockUnderMouse:', e);
        }
        return null; // Return null if no valid data is found
    }

        let itemData = null;
        
        if (currentView === 'treemap') {
            itemData = getBlockUnderMouse(chartX, chartY);
        } else if (currentView === 'sunburst') {
            // For sunburst, we need to use ECharts' built-in methods
            try {
                const hoveredElement = chartInstance.getZr().handler.findHover(chartX, chartY);
                if (hoveredElement && hoveredElement.target) {
                    // Try to extract data from sunburst element
                    const target = hoveredElement.target;
                    
                    // Check various ECharts internal properties for sunburst data
                    if (target.__ecData) {
                        itemData = target.__ecData;
                    } else {
                        // Look for data in internal properties
                        for (const prop of Object.keys(target)) {
                            if (prop.startsWith('__ec_inner_')) {
                                if (target[prop] && typeof target[prop].dataIndex === 'number') {
                                    const dataIndex = target[prop].dataIndex;
                                    const option = chartInstance.getOption();
                                    if (option.series && option.series[0] && option.series[0].data) {
                                        const candidateData = option.series[0].data[dataIndex];
                                        if (candidateData && candidateData.path) {
                                            itemData = candidateData;
                                            break;
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            } catch (error) {
                console.log('Error getting sunburst item data:', error);
            }
        }
        
        console.log('Item data found for context menu:', itemData);
        
        // Show context menu with the found item data
        showContextMenu(e.clientX, e.clientY, itemData);
    });

    // Hide context menu and search results
    document.addEventListener('click', (e) => {
        if (!contextMenu.contains(e.target)) {
            contextMenu.style.display = 'none';
        }
        if (!searchResultsContainer.contains(e.target) && e.target !== searchInput) {
            searchResultsContainer.style.display = 'none';
        }
    });

    // --- GLOBAL CALLBACK FUNCTIONS ---

    window.onDataChanged = function(changedDirs) {
        console.log("Live update received. Changes in:", changedDirs);
        statusBar.textContent = "Live update received, checking for refresh...";

        let needsRefresh = false;
        for (const dir of changedDirs) {
            // If the current path is inside a changed directory, we need to refresh.
            if (currentPath.startsWith(dir)) {
                needsRefresh = true;
                break;
            }
        }

        if (needsRefresh) {
            console.log("Current view is affected. Refreshing...");
            // Re-request the data for the current view
            if (currentView === 'treemap') {
                 if (treemapMode === 'structure') {
                    requestStructureView(currentPath);
                } else if (treemapMode === 'heatmap') {
                    // For heatmap, a simple refresh might not be enough, but this is a start
                    requestHeatmapView(currentPath, currentZoom);
                }
            } else if (currentView === 'sunburst') {
                 pywebview.api.get_sunburst_adaptive_view(currentPath, 4).then(data => {
                    if (data) renderView(data);
                });
            }
        } else {
             console.log("Current view is not affected. No refresh needed.");
             statusBar.textContent = "Live Mode Enabled. Monitoring for changes...";
        }
    };
        
    window.onQuickPreview = function(data) {
        console.log('Quick preview received:', data);
        isQuickPreview = true;
        chartInstance.hideLoading();
        chartInstance.showLoading({
            text: 'Quick preview loaded. Full scan in progress...',
            color: '#00aa00',
            maskColor: 'rgba(0, 0, 0, 0.5)'
        });
        
        if (data) {
            if (!data.children || !Array.isArray(data.children)) {
                data.children = [];
            }
            currentPath = data.path;
            currentRootPath = data.path;
            renderView(data);
            statusBar.textContent = 'Quick preview ready. Full scan continuing...';
        }
    };

    window.onScanComplete = function(data) {
        // This is now ONLY for cache loading
        console.log('Cache load complete:', data);
        navStack = [];
        currentZoom = 1.0;
        
        chartInstance.hideLoading();
        scanButton.disabled = false;
        cacheButton.disabled = false;
        driveSelect.disabled = false;
        liveModeToggle.disabled = false; // The critical fix
        
        if (data) {
            currentRootPath = data.path;
            currentPath = data.path;
            renderView(data);
            statusBar.textContent = `Cache loaded.`;
        } else {
            onCacheMiss();
        }
        isQuickPreview = false;
    };

window.onScanFinallyComplete = function() {
        console.log("Backend signaled full scan is complete. Now pulling final data...");
        statusBar.textContent = 'Scan complete. Fetching final data...';
        
        // Now we call the new API endpoint to get the data.
        pywebview.api.get_final_scan_data().then(data => {
            // This is the logic from the old onScanComplete function.
            console.log('Final scan data received:', data);
            navStack = [];
            currentZoom = 1.0;
            
            chartInstance.hideLoading(); // This will finally remove the "Quick preview..." message.
            scanButton.disabled = false;
            cacheButton.disabled = false;
            driveSelect.disabled = false;
            liveModeToggle.disabled = false; 
            
            if (data) {
                currentRootPath = data.path;
                currentPath = data.path;
                
                setTreemapMode('structure');
                renderView(data);
                statusBar.textContent = `Scan complete. Use Backspace to go up, Esc to reset`;
            } else {
                statusBar.textContent = 'Scan completed but failed to retrieve final data.';
            }
            isQuickPreview = false;
        }).catch(error => {
            console.error("Failed to pull final scan data:", error);
            onScanFailed("Error retrieving final scan results.");
        });
    };

    window.onCacheMiss = function() {
        statusBar.textContent = 'No cache found. Please perform a fresh scan.';
        scanButton.disabled = false;
        cacheButton.disabled = false;
        driveSelect.disabled = false;
        liveModeToggle.disabled = true;
        chartInstance.hideLoading();
    };

    window.onScanFailed = function(message = 'Scan failed. Please check permissions.') {
        console.error('Scan failed:', message);
        statusBar.textContent = message;
        scanButton.disabled = false;
        cacheButton.disabled = false;
        driveSelect.disabled = false;
        liveModeToggle.disabled = true; 
        chartInstance.hideLoading();
    };

    // Handle window resize
    window.addEventListener('resize', () => {
        chartInstance.resize();
    });

    // SPATIAL ZOOM: Keyboard shortcuts
    document.addEventListener('keydown', (e) => {
        if (e.target.matches('input, textarea')) return;

        // Spatial zoom shortcuts for heatmap mode
        if (currentView === 'treemap' && treemapMode === 'heatmap') {
            if (e.key === '-' || e.key === '_') {
                e.preventDefault();
                console.log(`[Spatial Zoom] Keyboard zoom out`);
                spatialZoomOut();
            }
        }

        if (e.key === 'Backspace') {
            e.preventDefault();
            // In the unified treemap, Backspace should act like a zoom out.
            if (currentView === 'treemap') {
                 const isRoot = currentPath.endsWith(':\\') && currentPath.length === 3;
                 if (currentPath && !isRoot) {
                    let parentPath = currentPath.substring(0, currentPath.lastIndexOf('\\'));
                    if (!parentPath.includes('\\')) parentPath += '\\';
                    
                    // FIX: Call the unified treemap request function.
                    currentZoom = 1.0;
                    requestTreemapView(parentPath, currentZoom);
                 }
            } else if (currentView === 'sunburst') {
                // Sunburst logic remains the same
                const isRoot = currentPath.endsWith(':\\') && currentPath.length === 3;
                 if (currentPath && !isRoot) {
                    let parentPath = currentPath.substring(0, currentPath.lastIndexOf('\\'));
                    if (!parentPath.includes('\\')) parentPath += '\\';
                    pywebview.api.get_sunburst_adaptive_view(parentPath, 4).then(data => {
                       if (data) renderView(data, true); // Use renderView
                    });
                 }
            }
        }

        if (e.key === 'ArrowLeft' && e.altKey) {
            e.preventDefault();
            if (navStack.length > 1) {
                navStack.pop(); // Remove current state
                const previousData = navStack[navStack.length - 1]; // Get previous state
                if (previousData) {
                    // FIX: Call the main renderView function, which dispatches correctly.
                    renderView(previousData, true);
                }
            }
        }

        if (e.key === 'Escape') {
            e.preventDefault();
            resetViewButton.click();
        }

        if (e.key === 'Tab' && !e.shiftKey && !e.ctrlKey) {
            e.preventDefault();
            chartToggle.checked = !chartToggle.checked;
            chartToggle.dispatchEvent(new Event('change'));
        }
    });

    // Initialize the application
    viewportSize = getViewportSize();
    statusBar.textContent = 'Ready. Select a drive and click Scan or Load Cache.';
    
    console.log('[Init] Spatial zoom implementation ready. Switch to Heatmap Mode and scroll over folders to zoom IN.');
});
